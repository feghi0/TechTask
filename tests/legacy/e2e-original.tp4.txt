import { test, expect } from '@playwright/test';
import { mkdirSync, writeFileSync } from 'node:fs';
const evidence='docs/entrega-tp4/03-evidencias';
test('crear, recargar, buscar y comprobar diseño móvil',async({page})=>{
  mkdirSync(evidence,{recursive:true});
  const errors:string[]=[];page.on('pageerror',e=>errors.push(e.message));page.on('console',message=>{if(message.type()==='error')errors.push(message.text());});
  await page.setViewportSize({width:1480,height:1000});
  await page.goto('/');await expect(page).toHaveTitle('TechTask — Mis tareas');
  await expect(page.getByRole('heading',{name:'Mis tareas',exact:true})).toBeVisible();
  await page.getByRole('button',{name:'Nueva tarea',exact:true}).click();
  const title=`Informe de redes ${Date.now()}`;
  await page.getByLabel('Título *').fill(title);await page.getByLabel('Materia *').fill('Redes');
  await page.getByLabel('Descripción').fill('Evidencia real de creación desde React.');
  await page.getByLabel('Marcar como favorita').check();
  const responsePromise=page.waitForResponse(r=>r.url().endsWith('/api/tasks')&&r.request().method()==='POST');
  await page.getByRole('button',{name:'Guardar tarea'}).click();
  const response=await responsePromise;expect(response.status()).toBe(201);
  writeFileSync(`${evidence}/creacion-http.json`,JSON.stringify({request:response.request().postDataJSON(),status:response.status(),response:await response.json()},null,2));
  await expect(page.getByRole('status')).toHaveText('Tarea guardada en el servidor.');
  await expect(page.getByRole('heading',{name:title})).toBeVisible();
  await page.screenshot({path:`${evidence}/desktop-formulario.png`,fullPage:true});
  await page.reload();await expect(page.getByRole('heading',{name:title})).toBeVisible();
  const list=await page.request.get('/api/tasks');expect(list.status()).toBe(200);expect((await list.json()).data.some((t:{title:string})=>t.title===title)).toBe(true);
  writeFileSync(`${evidence}/lectura-http.json`,JSON.stringify(await list.json(),null,2));
  await page.getByRole('textbox',{name:'Buscar tareas'}).fill('no existe esta coincidencia');await expect(page.getByRole('heading',{name:'Sin coincidencias'})).toBeVisible();
  await page.getByRole('textbox',{name:'Buscar tareas'}).fill(title);
  await page.getByLabel('Solo favoritas').check();await expect(page.getByRole('heading',{name:title})).toBeVisible();
  await page.screenshot({path:`${evidence}/desktop-recarga.png`,fullPage:true});
  await page.setViewportSize({width:390,height:844});
  await expect(page.getByRole('heading',{name:title})).toBeVisible();
  expect(await page.evaluate(()=>document.documentElement.scrollWidth<=window.innerWidth)).toBe(true);
  await page.screenshot({path:`${evidence}/mobile.png`,fullPage:true});
  expect(errors).toEqual([]);writeFileSync(`${evidence}/consola-navegador.json`,JSON.stringify(errors));
});
